searchBtn.addEventListener("click", () => {
  const yourLocation = document.getElementById("yourLocation").value.trim();
  const jobLocation = document.getElementById("jobLocation").value.trim();
  const jobKeyword = document.getElementById("jobKeyword").value.trim();
  const radius = document.getElementById("radius").value;

  let filtered = demoJobs;

  if (jobLocation) {
    filtered = filtered.filter(job =>
      job.location.toLowerCase().includes(jobLocation.toLowerCase())
    );
  }

  if (jobKeyword) {
    filtered = filtered.filter(job =>
      job.title.toLowerCase().includes(jobKeyword.toLowerCase()) ||
      job.snippet.toLowerCase().includes(jobKeyword.toLowerCase())
    );
  }

  renderResults(filtered, { yourLocation, jobLocation, jobKeyword, radius });
});

function renderResults(jobs, searchInfo) {
  if (jobs.length === 0) {
    resultsDiv.innerHTML = `
      <p class="empty">No demo jobs matched your search.</p>
      <p class="note">Try keywords like <strong>customer</strong>, <strong>designer</strong>, or <strong>intern</strong>.</p>
    `;
    jobDetailsDiv.innerHTML = `<p class="empty">Click a job card to view details here.</p>`;
    return;
  }

  resultsDiv.innerHTML = `
    <p class="note">
      Showing ${jobs.length} result(s) for
      <strong>${escapeHtml(searchInfo.jobKeyword || "any job")}</strong>
      in <strong>${escapeHtml(searchInfo.jobLocation || "any location")}</strong>
      (${escapeHtml(searchInfo.radius)} km)
    </p>
  `;

  jobs.forEach(job => {
    const card = document.createElement("div");
    card.className = "job-card";
    card.innerHTML = `
      <h3>${escapeHtml(job.title)}</h3>
      <div class="job-meta">
        ${escapeHtml(job.company)} • ${escapeHtml(job.location)} • ${escapeHtml(job.salary)}
      </div>
      <div class="job-snippet">${escapeHtml(job.snippet)}</div>
      <button data-job-id="${job.id}">View Job</button>
    `;
    resultsDiv.appendChild(card);
  });

  const buttons = resultsDiv.querySelectorAll("button[data-job-id]");
  buttons.forEach(btn => {
    btn.addEventListener("click", () => {
      const jobId = Number(btn.getAttribute("data-job-id"));
      const selectedJob = demoJobs.find(j => j.id === jobId);
      if (selectedJob) renderJobDetails(selectedJob);
    });
  });
}

function escapeHtml(str) {
  return String(str)
    .replaceAll("&", "&amp;")
    .replaceAll("<", "&lt;")
    .replaceAll(">", "&gt;")
    .replaceAll('"', "&quot;")
    .replaceAll("'", "&#039;");
}
